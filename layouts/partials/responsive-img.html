{{- /* Адаптивная картинка: srcset/sizes + WebP через Hugo Image Processing. Вход (dict): img — ресурс изображения (обязательно) sizes — атрибут sizes (обязательно) alt — alt-текст class — CSS-класс(ы), только для eager — bool: LCP (eager + fetchpriority="high"), по умолчанию false link — bool: вывести вместо (для ) widths — slice ширин, по умолчанию [480 800 1200 1600] quality — качество WebP, по умолчанию 78 */ -}} {{- $img := .img -}} {{- $widths := .widths | default (slice 480 800 1200 1600) -}} {{- $quality := .quality | default 78 -}} {{- $srcset := slice -}} {{- $first := "" -}} {{- $fallback := "" -}} {{- range $w := $widths -}} {{- if le $w $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q%d" $w $quality) -}} {{- $u := partial "cdn-url.html" $r.RelPermalink -}} {{- $srcset = $srcset | append (printf "%s %dw" $u $r.Width) -}} {{- if not $first }}{{ $first = $u }}{{ end -}} {{- if eq $w 800 }}{{ $fallback = $u }}{{ end -}} {{- end -}} {{- end -}} {{- if not $srcset -}} {{- /* Исходник уже 480px — отдаём как есть, единичным кандидатом. */ -}} {{- $first = partial "cdn-url.html" $img.RelPermalink -}} {{- $fallback = $first -}} {{- $srcset = slice (printf "%s %dw" $first $img.Width) -}} {{- end -}} {{- if not $fallback }}{{ $fallback = $first }}{{ end -}} {{- $src := $fallback -}} {{- $srcset = delimit $srcset ", " -}} {{- if .link -}} {{- /* Go html/template экранирует imagesrcset как обычный URL (пробелы → %20), а браузер разбирает imagesrcset по правилам srcset (литеральные пробелы). Обходим экранирование через safeHTML: все данные внутри — наши rel-permalink'и и статические sizes. */ -}} {{- $fp := "" -}} {{- if .eager }}{{ $fp = " fetchpriority=\"high\"" }}{{ end -}} {{- $link := printf "" $src $srcset .sizes $fp -}} {{- $link | safeHTML -}} {{- else -}} {{- end -}}
